```python
async def ws_scope(ws: WebSocket) -> Scope | None:
    """Which token the `?token=` query carries: the API token ("full"), the LCD device token ("device") or
```

- [ ] **Step 3: Create `server/devices.py`:**

```python
"""LCD kiosk presence: is a device-scoped client connected and reporting `device_health`?

Online = at least one device-token `/ws/events` socket and a `device_health` newer than DEVICE_STALE_S.
Every online/offline transition is emitted as `device_status` on /ws/events, so the iOS app and the agents
can show or react to "Mira's screen is offline".
"""

from __future__ import annotations

import asyncio
import time
from collections.abc import Callable
from typing import Any, Protocol

from server.logging_setup import get_logger

log = get_logger("devices")

DEVICE_STALE_S = 15.0  # the page sends device_health every 5 s; three misses = offline
DEVICE_CHECK_S = 5.0
DEVICE_NAME = "lcd"
# What a device-scoped client may send (only device_health) and which of its keys survive.
DEVICE_HEALTH_KEYS = frozenset({"type", "t", "app", "screen", "version", "battery_pct"})
# Event types delivered to a device-scoped /ws/events client: what the LCD renders, nothing else
# (no proposals, coach_adapt, notifications or engine diagnostics).
DEVICE_EVENT_TYPES = frozenset({"arrived", "greeting", "awaiting_gesture", "gesture", "countdown", "cue", "rep",
                                "set_done", "paused", "resumed", "stopped", "fall", "alert", "summary_ready",
                                "hr", "device_status", "ping"})


class Emitter(Protocol):
    def emit(self, event_type: str, payload: dict[str, Any]) -> None: ...


class DeviceRegistry:
    def __init__(self, hub: Emitter, *, stale_s: float = DEVICE_STALE_S,
                 clock: Callable[[], float] = time.time) -> None:
        self._hub = hub
        self._stale_s = stale_s
        self._clock = clock
        self._connections = 0
        self._last_seen: float | None = None
        self._last_health: dict[str, Any] | None = None
        self._task: asyncio.Task[None] | None = None
        self.online = False

    def connected(self) -> None:
        self._connections += 1
        self.check()

    def disconnected(self) -> None:
        self._connections = max(0, self._connections - 1)
        self.check()

    def on_health(self, msg: dict[str, Any]) -> None:
        self._last_seen = self._clock()
        self._last_health = {k: v for k, v in msg.items() if k != "type"}
        self.check()

    def check(self) -> None:
        fresh = self._last_seen is not None and self._clock() - self._last_seen < self._stale_s
        online = self._connections > 0 and fresh
        if online == self.online:
            return
        self.online = online
        log.info("lcd online" if online else "lcd offline", extra={"connections": self._connections})
        self._hub.emit("device_status", {"device": DEVICE_NAME, "online": online, "t": round(self._clock(), 3),
                                         "last_health": self._last_health})

    def status(self) -> dict[str, Any]:
        return {"online": self.online, "connections": self._connections, "last_seen": self._last_seen,
                "last_health": self._last_health}

    async def run(self) -> None:
        while True:
            await asyncio.sleep(DEVICE_CHECK_S)
            self.check()

    def start(self) -> None:
        self._task = asyncio.create_task(self.run(), name="device-registry")

    async def stop(self) -> None:
        if self._task is not None:
            self._task.cancel()
            try:
                await self._task
            except asyncio.CancelledError:
                pass
```

```js
    live: null,
    arrived: false,
    awaitingGesture: false,
    lastCorrectionAt: -Infinity,
    lastGoodRepAt: -Infinity,
    alertOpen: false,
    eventCaption: "",
    socket: "connecting",
    closedAt: null,
  };
}

export function reduce(state, input, now) {
  const { channel, msg } = input;
  if (!msg || typeof msg !== "object") return state;
  if (channel === "socket") {
    if (msg.status === "open") return { ...state, socket: "open", closedAt: null };
    if (msg.status === "rejected") return { ...state, socket: "rejected" };
    if (msg.status === "closed") return { ...state, socket: "closed", closedAt: state.closedAt ?? now };
    return state;
  }
  if (channel === "live") {
    let next = { ...state, live: msg };
    if (msg.phase === "countdown" || msg.phase === "active") next = { ...next, awaitingGesture: false };
    return next;
  }
  switch (msg.type) {
    case "arrived":
      return { ...state, arrived: true, awaitingGesture: false, alertOpen: false, eventCaption: "" };
    case "greeting":
      return { ...state, arrived: true, eventCaption: textOf(msg.text, state.eventCaption) };
    case "awaiting_gesture":
      return { ...state, arrived: true, awaitingGesture: true };
    case "gesture":
      return { ...state, awaitingGesture: false };
    case "cue": {
      const correction = !NON_CORRECTION_CUES.has(msg.id);
      return { ...state, eventCaption: textOf(msg.text, state.eventCaption),
        lastCorrectionAt: correction ? now : state.lastCorrectionAt,
        alertOpen: msg.id === "check_in" ? true : state.alertOpen };
    }
    case "rep":
      return msg.quality === "good" ? { ...state, lastGoodRepAt: now } : state;
    case "summary_ready":
      return { ...state, eventCaption: textOf(msg.spoken, state.eventCaption) };
    case "fall":
      return { ...state, alertOpen: true };
    case "alert":
      return { ...state, alertOpen: !CLOSED_ALERT_STEPS.has(msg.step) };
    case "stopped":
      return { ...state, awaitingGesture: false };
    default:
      return state;
  }
}

function textOf(value, fallback) {
  return typeof value === "string" && value.trim() ? value.trim() : fallback;
}

function num(value) {
  return typeof value === "number" && Number.isFinite(value) ? value : null;
}

export function view(state, now) {
  const live = state.live ?? {};
  const phase = live.phase ?? "idle";
  const repsTarget = num(live.reps_target);
  const leftPct = num(live.load_l_pct);
  const balance = leftPct === null ? null
    : { leftPct: Math.round(leftPct), ok: Math.abs(leftPct - 50) <= BALANCE_LIMIT };
  const set = num(live.set);
  const setsTarget = num(live.sets_target);
  const liveCaption = live.caption && typeof live.caption.text === "string" ? live.caption.text : "";
  const base = {
    offline: state.socket === "closed" && state.closedAt !== null && now - state.closedAt > OFFLINE_AFTER_MS,
    setLabel: set !== null && setsTarget !== null ? `SET ${set} OF ${setsTarget}` : null,
    hr: num(live.hr),
    firstName: typeof live.first_name === "string" ? live.first_name : "",
    exerciseName: typeof live.exercise_name === "string" ? live.exercise_name : "",
    dose: setsTarget !== null && repsTarget !== null ? `${setsTarget} sets · ${repsTarget} reps` : "",
    rep: num(live.rep) ?? 0,
    repsTarget,
    countdown: num(live.countdown),
    balance,
    // The newest line wins: an event caption is at least as new as the 10 Hz snapshot; live.caption repaints
    // a kiosk that (re)joined mid-session before any event arrived.
    caption: state.eventCaption || liveCaption,
    summary: null,
  };

  if (state.socket === "rejected") return { ...base, screen: "unpaired", mood: null };
  if (live.incident || phase === "incident" || state.alertOpen) return { ...base, screen: "fall", mood: null };
  if (live.paused || phase === "paused") return { ...base, screen: "paused", mood: "sleeping" };
  if (phase === "countdown") return { ...base, screen: "countdown", mood: null };
  if (phase === "rest" || phase === "done") {
    const summary = { reps: base.rep, repsTarget, balance: balance
      ? `${balance.leftPct} / ${100 - balance.leftPct}` : null };
    return { ...base, screen: "setDone", mood: "celebrating", summary };
  }
  if (state.awaitingGesture) return { ...base, screen: "gesture", mood: "idle" };
  if (phase === "active") {
    const correcting = now - state.lastCorrectionAt <= CUE_WINDOW_MS || (balance !== null && !balance.ok);
    if (correcting) return { ...base, screen: "cue", mood: "worried" };
    const cheering = now - state.lastGoodRepAt <= CHEER_MS;
    return { ...base, screen: "rep", mood: cheering ? "cheering" : "idle" };
  }
  if (state.arrived) return { ...base, screen: "greeting", mood: "idle" };
  return { ...base, screen: "idle", mood: "idle" };
}
```

```js
export const TOKEN_KEY = "mira.deviceToken";

export function readToken(location, storage, history) {
  const params = new URLSearchParams((location.hash || "").replace(/^#/, ""));
  const fromHash = params.get("token");
  if (fromHash) {
    try { storage.setItem(TOKEN_KEY, fromHash); } catch { /* private mode: keep it for this load only */ }
    history.replaceState(null, "", location.pathname + (location.search || ""));
    return fromHash;
  }
  try { return storage.getItem(TOKEN_KEY); } catch { return null; }
}

export function wsUrl(location, path, token) {
```

```js
// One WebSocket that reconnects with backoff (1, 2, 4 … 30 s + jitter), like the iOS app (LiveSocket.swift).
// Server pings are ignored. A policy close (1008) after open means "rejected": stop retrying. A token refused at
// the handshake looks like any network failure to a browser (close 1006), so it shows as "Reconnecting…".

const MAX_BACKOFF_MS = 30000;

export class ReconnectingSocket {
  constructor(url, { onMessage, onStatus, WebSocketImpl = globalThis.WebSocket, random = Math.random } = {}) {
    this.url = url;
    this.onMessage = onMessage;
    this.onStatus = onStatus;
    this.WebSocketImpl = WebSocketImpl;
    this.random = random;
    this.attempt = 0;
    this.ws = null;
    this.timer = null;
    this.stopped = false;
  }

  start() {
    this.stopped = false;
    this.connect();
  }

  connect() {
    const ws = new this.WebSocketImpl(this.url);
    this.ws = ws;
    ws.onopen = () => {
      this.attempt = 0;
      this.onStatus?.("open");
    };
    ws.onmessage = (event) => {
      let msg;
      try { msg = JSON.parse(event.data); } catch { return; }
      if (!msg || typeof msg !== "object" || msg.type === "ping") return;
      this.onMessage?.(msg);
    };
    ws.onclose = (event) => {
      if (this.stopped) return;
      if (event.code === 1008) {
        this.onStatus?.("rejected");
        return;
      }
      this.onStatus?.("closed");
      this.schedule();
    };
    ws.onerror = () => { /* onclose follows */ };
  }

  schedule() {
    const base = Math.min(MAX_BACKOFF_MS, 1000 * 2 ** this.attempt);
    this.attempt += 1;
    this.timer = setTimeout(() => this.connect(), base + this.random() * 500);
  }

  send(obj) {
    if (this.ws && this.ws.readyState === 1) this.ws.send(JSON.stringify(obj));
  }

  stop() {
    this.stopped = true;
    clearTimeout(this.timer);
    this.ws?.close();
  }
}
```

```js
import { readToken, wsUrl } from "./config.js";
import { ReconnectingSocket } from "./socket.js";
import { initialState, reduce, view } from "./state.js";
import { render } from "./view.js";
import { AudioPlayer } from "./audio.js";
import { loadMascot } from "./mascot.js";

const VERSION = "1";
const HEALTH_EVERY_MS = 5000;
const AUDIO_EVENTS = new Set(["greeting", "cue", "summary_ready"]);

function fitStage() {
  const scale = Math.min(window.innerWidth / 1024, window.innerHeight / 600);
  document.documentElement.style.setProperty("--stage-scale", String(scale));
}

async function boot() {
  fitStage();
  window.addEventListener("resize", fitStage);
  const mascot = await loadMascot();
  let state = initialState();
  const draw = () => render(document, view(state, Date.now()), mascot);
  const dispatch = (input) => { state = reduce(state, input, Date.now()); draw(); };

  const token = readToken(window.location, window.localStorage, window.history);
  if (!token) {
    dispatch({ channel: "socket", msg: { status: "rejected" } });
    return;
  }
  const audio = new AudioPlayer();
  const status = (s) => dispatch({ channel: "socket", msg: { status: s } });
  const live = new ReconnectingSocket(wsUrl(window.location, "/ws/live", token), {
    onMessage: (msg) => dispatch({ channel: "live", msg }),
  });
  const health = () => events.send({ type: "device_health", app: "lcd", t: Date.now() / 1000,
    screen: `${window.innerWidth}x${window.innerHeight}`, version: VERSION });
  const events = new ReconnectingSocket(wsUrl(window.location, "/ws/events", token), {
    onStatus: (s) => { status(s); if (s === "open") health(); },
    onMessage: (msg) => {
      dispatch({ channel: "events", msg });
      if (AUDIO_EVENTS.has(msg.type)) audio.play(msg.audio_url);
    },
  });
  live.start();
  events.start();
  setInterval(health, HEALTH_EVERY_MS);
  setInterval(draw, 250); // time-based states (cheer, cue window, offline grace) expire without new messages
  draw();
}
```
